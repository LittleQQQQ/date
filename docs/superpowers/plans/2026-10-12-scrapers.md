```python
"""抓網頁的共用工具：限速、重試、快取。

- 同一個網站每秒最多一次請求（min_interval），避免造成對方負擔。
- 連線失敗、429、5xx 會等一下再試（2 秒、4 秒），最多 retries 次；其他 4xx 直接失敗。
- cache_days > 0 的網址存在 data/cache/，有效期間內不重抓（地圖頁、電影頁這種很少變的頁面）。
"""
from __future__ import annotations

import hashlib
import time
from collections.abc import Callable
from pathlib import Path
from urllib.parse import urlsplit

import httpx

from ..config import ROOT_DIR

CACHE_DIR = ROOT_DIR / "data" / "cache"
USER_AGENT = "date-planner/0.1 (student project; +https://github.com/LittleQQQQ/date)"
RETRY_STATUS = {429, 500, 502, 503, 504}


class FetchError(Exception):
    """抓不到這一頁（重試後仍失敗，或網站回 4xx）。"""


class Fetcher:
    def __init__(
        self,
        cache_dir: Path | None = CACHE_DIR,
        min_interval: float = 1.0,
        retries: int = 3,
        timeout: float = 30.0,
        client: httpx.Client | None = None,
        sleep: Callable[[float], None] = time.sleep,
        clock: Callable[[], float] = time.monotonic,
    ):
        self.cache_dir = cache_dir
        self.min_interval = min_interval
        self.retries = retries
        self.client = client or httpx.Client(
            timeout=timeout, headers={"User-Agent": USER_AGENT}, follow_redirects=True
        )
        self.sleep = sleep
        self.clock = clock
        self._last: dict[str, float] = {}

    def get(self, url: str, cache_days: float = 0) -> str:
        if cache_days > 0 and (cached := self._read_cache(url, cache_days)) is not None:
            return cached
        text = self._request("GET", url)
        if cache_days > 0:
            self._write_cache(url, text)
        return text

    def post(self, url: str, data: dict[str, str]) -> str:
        return self._request("POST", url, data=data)

    def _request(self, method: str, url: str, **kwargs) -> str:
        last_error = ""
        for attempt in range(self.retries):
            self._wait_turn(url)
            try:
                resp = self.client.request(method, url, **kwargs)
            except httpx.TransportError as exc:
                last_error = f"{type(exc).__name__}: {exc}"
            else:
                if resp.status_code < 400:
                    return resp.text
                if resp.status_code not in RETRY_STATUS:
                    raise FetchError(f"{url} 回應 {resp.status_code}")
                last_error = f"回應 {resp.status_code}"
            if attempt < self.retries - 1:
                self.sleep(2 * 2**attempt)
        raise FetchError(f"{url} 重試 {self.retries} 次仍失敗：{last_error}")

    def _wait_turn(self, url: str) -> None:
        host = urlsplit(url).netloc
        if host in self._last:
            gap = self.min_interval - (self.clock() - self._last[host])
            if gap > 0:
                self.sleep(gap)
        self._last[host] = self.clock()

    def _cache_path(self, url: str) -> Path:
        return self.cache_dir / f"{hashlib.sha1(url.encode('utf-8')).hexdigest()}.html"

    def _read_cache(self, url: str, days: float) -> str | None:
        if self.cache_dir is None:
            return None
        path = self._cache_path(url)
        if not path.exists() or time.time() - path.stat().st_mtime > days * 86400:
            return None
        return path.read_text(encoding="utf-8")

    def _write_cache(self, url: str, text: str) -> None:
        if self.cache_dir is None:
            return
        self.cache_dir.mkdir(parents=True, exist_ok=True)
        self._cache_path(url).write_text(text, encoding="utf-8")
```

```python
"""測試用的假抓網頁工具：網址對到樣本，不連網。"""
from __future__ import annotations

from collections.abc import Callable
from pathlib import Path
from typing import Union

from app.collectors.http import FetchError

SAMPLES = Path(__file__).parent / "samples"

PageSource = Union[str, Path, Exception, Callable[[dict | None], str], list]


class FakeFetcher:
    """pages：網址 → 內容。內容可以是字串、樣本檔路徑、要丟出的例外、
    callable(data)（POST 時拿到表單資料），或上述元素的清單（每次呼叫取出第一個）。"""

    def __init__(self, pages: dict[str, PageSource]):
        self.pages = pages
        self.requested: list[str] = []
        self.posts: list[dict] = []

    def get(self, url: str, cache_days: float = 0) -> str:
        return self._serve(url, None)

    def post(self, url: str, data: dict[str, str]) -> str:
        self.posts.append(data)
        return self._serve(url, data)

    def _serve(self, url: str, data: dict | None) -> str:
        self.requested.append(url)
        page = self.pages.get(url)
        if isinstance(page, list):
            page = page.pop(0) if page else None
        if page is None:
            raise FetchError(f"{url} 回應 404")
        if isinstance(page, Exception):
            raise page
        if isinstance(page, Path):
            return page.read_text(encoding="utf-8")
        if callable(page):
            return page(data)
        return page
```

```python
同時離好幾個商圈都夠近時，取走路最近的。
"""
from __future__ import annotations

from dataclasses import dataclass, field

from sqlalchemy import select
from sqlalchemy.orm import Session

from ..geo import walk_minutes
from ..models import Area, Cinema, Station
from ..planner import STATION_WALK_MAX


@dataclass(frozen=True)
class Point:
    lat: float
    lng: float


@dataclass
class AreaCoverage:
    id: str
    city: str
    stations: list[Point]
    cinemas: list[Point] = field(default_factory=list)

    def walk_from_station(self, p: Point) -> int:
        return min(walk_minutes(s, p) for s in self.stations)

    def walk_to_cinemas(self, p: Point) -> int:
        """到最近影城的步行分鐘數；還沒有影城資料時改用最近的車站。"""
        return min(walk_minutes(t, p) for t in (self.cinemas or self.stations))


@dataclass
class Coverage:
    areas: list[AreaCoverage]

    def cities(self) -> set[str]:
        return {a.city for a in self.areas}

    def area_for(self, lat: float, lng: float) -> str | None:
        p = Point(lat, lng)
        best = min(((a.walk_from_station(p), a.id) for a in self.areas if a.stations), default=None)
        return best[1] if best and best[0] <= STATION_WALK_MAX else None

    def get(self, area_id: str) -> AreaCoverage:
        return next(a for a in self.areas if a.id == area_id)


def load_coverage(session: Session) -> Coverage:
    areas = []
    for area in session.scalars(select(Area).order_by(Area.id)):
        stations = [Point(s.lat, s.lng) for s in session.scalars(select(Station).where(Station.area_id == area.id))]
        if not stations:
            continue
        cinemas = [Point(c.lat, c.lng) for c in session.scalars(select(Cinema).where(Cinema.area_id == area.id))]
        areas.append(AreaCoverage(area.id, area.city, stations, cinemas))
    return Coverage(areas)
```

```python
def upsert_batch(session: Session, source: str, batch: Batch) -> dict[str, int]:
    now = now_tw()
    counts: dict[str, int] = {}

    for model, records in ((Area, batch.areas), (Station, batch.stations)):
        for rec in records:
            row = session.get(model, rec.id) or model(id=rec.id)
            for field, value in rec.model_dump().items():
                setattr(row, field, value)
            row.source, row.fetched_at = source, now
            session.add(row)
        session.flush()
    counts.update(areas=len(batch.areas), stations=len(batch.stations))

    for rec in batch.transit_links:
        row = session.scalar(
            select(TransitLink).where(
                TransitLink.from_station_id == rec.from_station_id,
                TransitLink.to_station_id == rec.to_station_id,
                TransitLink.line == rec.line,
            )
        ) or TransitLink()
        for field, value in rec.model_dump().items():
            setattr(row, field, value)
        row.source, row.fetched_at = source, now
        session.add(row)
    counts["transit_links"] = len(batch.transit_links)

    # 影城、餐廳要屬於資料庫裡有的商圈；爬蟲已經篩過，這裡再擋一次
    area_ids = set(session.scalars(select(Area.id)))
    outside = 0
    cinema_rows: dict[str, Cinema] = {}
    for rec in batch.cinemas:
        if rec.area_id not in area_ids:
            outside += 1
            continue
        cinema_rows[rec.external_id] = _upsert(session, Cinema, source, rec, now)
    movie_rows = {rec.external_id: _upsert(session, Movie, source, rec, now) for rec in batch.movies}
    seen_restaurants: set[str] = set()
    for rec in batch.restaurants:
        if rec.area_id not in area_ids:
            outside += 1
            continue
        _upsert(session, Restaurant, source, rec, now)
        seen_restaurants.add(rec.external_id)
    session.flush()

    removed = 0
    if batch.restaurant_areas:
        stale = select(Restaurant).where(
            Restaurant.source == source,
            Restaurant.area_id.in_(batch.restaurant_areas),
            Restaurant.external_id.not_in(seen_restaurants),
        )
        for row in session.scalars(stale).all():
            session.delete(row)
            removed += 1
    counts.update(
        cinemas=len(cinema_rows), movies=len(movie_rows), restaurants=len(seen_restaurants),
        restaurants_removed=removed, places_outside_areas=outside,
    )
    session.flush()

    if batch.showtime_dates:
        first, last = batch.showtime_dates
        stmt = delete(Showtime).where(
            Showtime.source == source,
            Showtime.date >= first.isoformat(),
            Showtime.date <= last.isoformat(),
        )
        if batch.showtime_cinemas is not None:
            ids = [
                row.id
                for eid in batch.showtime_cinemas
                if (row := cinema_rows.get(eid) or _find(session, Cinema, source, eid))
            ]
            stmt = stmt.where(Showtime.cinema_id.in_(ids))
        session.execute(stmt)

    # 對不到電影編號時，用正規化後的片名找同一來源的電影
    movies_by_title = {
        normalize_title(m.title): m.id for m in session.scalars(select(Movie).where(Movie.source == source))
    }
    added = skipped = unlinked = 0
    for rec in batch.showtimes:
        cinema = cinema_rows.get(rec.cinema_external_id) or _find(session, Cinema, source, rec.cinema_external_id)
        if cinema is None:
            skipped += 1
            continue
        movie = None
        if rec.movie_external_id:
            movie = movie_rows.get(rec.movie_external_id) or _find(session, Movie, source, rec.movie_external_id)
        movie_id = movie.id if movie else movies_by_title.get(normalize_title(rec.raw_title))
        unlinked += movie_id is None
        session.add(
            Showtime(
                source=source,
                external_id=rec.external_id,
                cinema_id=cinema.id,
                movie_id=movie_id,
                raw_title=rec.raw_title,
                date=rec.date.isoformat(),
                start=rec.start,
                label=rec.label,
                price=rec.price,
                source_url=rec.source_url,
                estimated=rec.estimated,
                fetched_at=now,
            )
        )
        added += 1
    counts.update(showtimes=added, showtimes_skipped=skipped, showtimes_without_movie=unlinked)
    counts.update(batch.stats)
    session.flush()
    return counts
```

```python
def precompute_jev(session: Session) -> int:
    """事先問好不會隨使用者輸入改變的題目（餐廳、場次標示、片型），回傳實際呼叫 Jev 的次數。"""
    jev = JevService(session)
    for r in session.scalars(select(Restaurant)):
        for question in (DATE_FIT, MEAL_PACE, CUISINE):
            jev.ask(question, restaurant_subject(r), restaurant_context(r))
    for label in session.scalars(select(Showtime.label).distinct()):
        for question in (SCREEN, AUDIO):
            jev.ask(question, label_subject(label), {"label": label})
    # 來源沒給片型的電影（開眼）由 Jev 判斷；沒把握就留空
    for m in session.scalars(select(Movie).where(Movie.genre == "")):
        answer = jev.ask(GENRE, movie_subject(m), movie_context(m))
        if answer.confidence >= settings.jev_confidence_threshold:
            m.genre = answer.top
    session.flush()
    return jev.calls
```

```python
"""開眼電影網爬蟲：影城、電影、場次。

一個網站涵蓋全台幾乎所有影城。只抓有商圈的縣市，而且只留走得到商圈車站的影城。
- 地區頁 /showtime/{地區}/              → 影城清單
- 地圖頁 GMap2018.cfm?tid={影城}         → 經緯度（快取 30 天）
- 時刻表 /showtime/{影城}/{地區}/[日期]/ → 地址、每部片的片長、版本、場次（今天起 4 天）
- 電影頁 /movie/{電影}/                  → 劇情簡介（快取 30 天）
網站沒有票價和片型：票價依版本估計（標成估計），片型交給 Jev 判斷。
"""
from __future__ import annotations

import datetime as dt
import re
from dataclasses import dataclass, field

from bs4 import BeautifulSoup, Comment

BASE_URL = "https://www.atmovies.com.tw"
MAP_URL = "http://ww2.atmovies.com.tw/movie/GMap2018.cfm?tid={tid}"
SITE_DAYS = 4  # 網站提供今天起幾天的時刻表
CACHE_DAYS = 30  # 地圖頁、電影頁很少變
SYNOPSIS_MAX = 600
DEFAULT_LABEL = "數位"  # 沒標版本的場次
PREMIUM_WORDS = ("IMAX", "4DX", "巨幕", "TITAN", "SCREENX", "DOLBY")
PRICE_PREMIUM, PRICE_NORMAL = 420, 300

# 縣市 → 開眼的地區代碼（2026-10-12 從網站確認；連江縣沒有）
REGION_BY_CITY = {
    "基隆市": "a01", "台北市": "a02", "新北市": "a02", "桃園市": "a03",
    "新竹市": "a35", "新竹縣": "a35", "苗栗縣": "a37", "台中市": "a04",
    "彰化縣": "a47", "雲林縣": "a45", "南投縣": "a49", "嘉義市": "a05",
    "嘉義縣": "a05", "台南市": "a06", "高雄市": "a07", "屏東縣": "a87",
    "宜蘭縣": "a39", "花蓮縣": "a38", "台東縣": "a89", "澎湖縣": "a69",
    "金門縣": "a68",
}


@dataclass(frozen=True)
class TheaterLink:
    tid: str
    name: str


@dataclass
class FilmShowing:
    movie_id: str
    title: str
    runtime_min: int | None  # 抓不到片長時為 None，這部片的場次會跳過
    version: str  # 「國語版」「4DX版」；沒標為空字串
    times: list[str] = field(default_factory=list)


@dataclass
class TheaterDay:
    date: dt.date | None
    address: str
    films: list[FilmShowing]


_THEATER_HREF = re.compile(r"^/showtime/(t\w+)/(a\d+)/$")
_COORDS = re.compile(r"lat:\s*(-?\d+(?:\.\d+)?)\s*,\s*lng:\s*(-?\d+(?:\.\d+)?)")
_DATE = re.compile(r"(\d{4})/(\d{1,2})/(\d{1,2})")
_ADDRESS = re.compile(r"地址\s*[:：]\s*([^<\n(]+)")
_MOVIE_HREF = re.compile(r"^/movie/([^/]+)/$")
_RUNTIME = re.compile(r"片長：\s*(\d+)\s*分")
_TIME = re.compile(r"^(\d{1,2})[：:](\d{2})$")


def region_url(region: str) -> str:
    return f"{BASE_URL}/showtime/{region}/"


def theater_url(tid: str, region: str, day: dt.date, today: dt.date) -> str:
    base = f"{BASE_URL}/showtime/{tid}/{region}/"
    return base if day == today else f"{base}{day:%Y%m%d}/"


def movie_url(movie_id: str) -> str:
    return f"{BASE_URL}/movie/{movie_id}/"


def parse_region(html: str, region: str) -> list[TheaterLink]:
    soup = BeautifulSoup(html, "html.parser")
    found: dict[str, TheaterLink] = {}
    for a in soup.find_all("a", href=True):
        m = _THEATER_HREF.match(a["href"])
        if m and m.group(2) == region and m.group(1) not in found:
            found[m.group(1)] = TheaterLink(m.group(1), a.get_text(strip=True))
    return list(found.values())


def parse_map(html: str) -> tuple[float, float] | None:
    m = _COORDS.search(html)
    return (float(m.group(1)), float(m.group(2))) if m else None


def normalize_time(text: str) -> str | None:
    """「10：20」→「10:20」；凌晨的場次算前一天的深夜場，「00：40」→「24:40」。"""
    m = _TIME.match(text.strip())
    if not m:
        return None
    hour, minute = int(m.group(1)), int(m.group(2))
    if hour < 5:
        hour += 24
    return f"{hour:02d}:{minute:02d}"


def parse_theater_day(html: str) -> TheaterDay:
    soup = BeautifulSoup(html, "html.parser")
    day = None
    for h3 in soup.find_all("h3"):
        if m := _DATE.search(h3.get_text()):
            day = dt.date(int(m.group(1)), int(m.group(2)), int(m.group(3)))
            break
    # 地址那一行的 <li> 在原始網頁裡沒有關閉，直接從原始碼抓
    address = m.group(1).strip() if (m := _ADDRESS.search(html)) else ""

    films = []
    for block in soup.select("ul#theaterShowtimeTable"):
        link = block.select_one("li.filmTitle a[href]")
        movie = _MOVIE_HREF.match(link["href"]) if link else None
        if not movie:
            continue  # 沒有電影頁的特別活動，不知道片長
        runtime = _RUNTIME.search(block.get_text())
        minutes = int(runtime.group(1)) if runtime else 0
        version = block.select_one("li.filmVersion")
        times = [t for li in block.find_all("li") if (t := normalize_time(li.get_text(strip=True)))]
        films.append(
            FilmShowing(
                movie_id=movie.group(1),
                title=link.get_text(strip=True),
                runtime_min=minutes or None,
                version=version.get_text(strip=True) if version else "",
                times=times,
            )
        )
    return TheaterDay(day, address, films)


def parse_synopsis(html: str) -> str:
    start, end = html.find("<!-- Story info start -->"), html.find("<!-- Story info end -->")
    if start < 0 or end < start:
        return ""
    soup = BeautifulSoup(html[start:end], "html.parser")
    for comment in soup.find_all(string=lambda s: isinstance(s, Comment)):
        comment.extract()
    for br in soup.find_all("br"):
        br.replace_with("\n")
    text = re.sub(r"\s+", " ", soup.get_text()).strip()
    return text.removeprefix("劇情簡介").strip()[:SYNOPSIS_MAX]


def price_for(label: str) -> int:
    return PRICE_PREMIUM if any(w in label.upper() for w in PREMIUM_WORDS) else PRICE_NORMAL
```

```python
class AtmoviesCollector(Collector):
    name = "atmovies"
    needs_coverage = True

    def __init__(self, coverage: Coverage, fetcher: Fetcher | None = None, today: dt.date | None = None):
        self.coverage = coverage
        self.fetcher = fetcher or Fetcher()
        self.today = today

    def collect(self, start: dt.date, days: int) -> Batch:
        today = self.today or today_tw()
        last_site_day = today + dt.timedelta(days=SITE_DAYS - 1)
        dates = [d for d in (start + dt.timedelta(days=i) for i in range(days)) if today <= d <= last_site_day]
        regions = sorted({REGION_BY_CITY[c] for c in self.coverage.cities() if c in REGION_BY_CITY})
        if not dates or not regions:
            return Batch()

        self.failed = self.skipped = 0
        cinemas: list[CinemaRecord] = []
        films: dict[str, FilmShowing] = {}  # 電影編號 → 第一次看到的片名、片長
        showtimes: list[ShowtimeRecord] = []
        for region in regions:
            theaters = parse_region(self.fetcher.get(region_url(region)), region)
            if not theaters:
                raise ScrapeError(f"開眼地區頁 {region} 解析不到影城，網站可能改版了")
            for theater in theaters:
                result = self._theater(theater, region, dates, today)
                if result is None:
                    continue
                cinema, theater_films, theater_showtimes = result
                cinemas.append(cinema)
                for film in theater_films:
                    films.setdefault(film.movie_id, film)
                showtimes += theater_showtimes

        movies = [self._movie(film) for film in films.values()]
        return Batch(
            cinemas=cinemas,
            movies=movies,
            showtimes=showtimes,
            showtime_dates=(dates[0], dates[-1]),
            showtime_cinemas=[c.external_id for c in cinemas],
            stats={"failed_pages": self.failed, "showtimes_skipped_no_runtime": self.skipped},
        )

    def _theater(
        self, theater: TheaterLink, region: str, dates: list[dt.date], today: dt.date
    ) -> tuple[CinemaRecord, list[FilmShowing], list[ShowtimeRecord]] | None:
        """抓一家影城。不在商圈範圍、或任何一頁抓失敗就回傳 None（保留這家影城上次的場次）。"""
        try:
            point = parse_map(self.fetcher.get(MAP_URL.format(tid=theater.tid), cache_days=CACHE_DAYS))
        except FetchError:
            point = None
        if point is None:
            self.failed += 1
            return None
        area = self.coverage.area_for(*point)
        if area is None:
            return None

        pages = []
        for day in dates:
            url = theater_url(theater.tid, region, day, today)
            try:
                page = parse_theater_day(self.fetcher.get(url))
            except FetchError:
                self.failed += 1
                return None
            if page.date != day:  # 網站正在換日或頁面不對，這家這次先不更新
                self.failed += 1
                return None
            pages.append((day, url, page))

        cinema = CinemaRecord(
            external_id=theater.tid, name=theater.name, area_id=area, address=pages[0][2].address,
            lat=point[0], lng=point[1], source_url=theater_url(theater.tid, region, today, today),
        )
        films: list[FilmShowing] = []
        showtimes: dict[str, ShowtimeRecord] = {}
        for day, url, page in pages:
            for film in page.films:
                if film.runtime_min is None:
                    self.skipped += len(film.times)
                    continue
                films.append(film)
                label = film.version or DEFAULT_LABEL
                for start in film.times:
                    eid = f"{theater.tid}|{film.movie_id}|{label}|{day.isoformat()}|{start}"
                    showtimes[eid] = ShowtimeRecord(
                        external_id=eid, cinema_external_id=theater.tid, movie_external_id=film.movie_id,
                        raw_title=film.title, date=day, start=start, label=label, price=price_for(label),
                        source_url=url, estimated=["price"],
                    )
        return cinema, films, list(showtimes.values())

    def _movie(self, film: FilmShowing) -> MovieRecord:
        url = movie_url(film.movie_id)
        try:
            synopsis = parse_synopsis(self.fetcher.get(url, cache_days=CACHE_DAYS))
        except FetchError:
            self.failed += 1
            synopsis = ""
        return MovieRecord(
            external_id=film.movie_id, title=film.title, runtime_min=film.runtime_min,
            synopsis=synopsis, source_url=url,
        )
```

```python
"""OpenStreetMap 餐廳爬蟲（透過 Overpass API）。

OSM 有店名、座標，部分店家有料理類型（cuisine）與營業時間（opening_hours）；沒有評論和價位。

    "cafe": ("10:00", "19:00"),

ESTIMATED_PRICE = {"restaurant": 500, "cafe": 300, "fast_food": 200, "bar": 700, "pub": 700}
PRICEY_CATEGORIES = ("火鍋／燒肉", "義式／西式")

_DAYS = ("Mo", "Tu", "We", "Th", "Fr", "Sa", "Su")
_SPAN_TEXT = r"\d{1,2}:\d{2}\s*-\s*\d{1,2}:\d{2}"
_RULE = re.compile(
    rf"^(?:(?P<days>[A-Za-z][A-Za-z,\- ]*?)\s+)?(?P<times>off|closed|{_SPAN_TEXT}(?:\s*,\s*{_SPAN_TEXT})*)$"
)
_SPAN = re.compile(r"(\d{1,2}):(\d{2})\s*-\s*(\d{1,2}):(\d{2})")


def _parse_days(text: str) -> list[int] | None:
    days: list[int] = []
    for part in text.replace(" ", "").split(","):
        if part == "PH":  # 國定假日的規則忽略
            continue
        if "-" in part:
            first, last = part.split("-", 1)
            if first not in _DAYS or last not in _DAYS:
                return None
            i, j = _DAYS.index(first), _DAYS.index(last)
            days += [(i + k) % 7 for k in range((j - i) % 7 + 1)]  # 可以跨週，例如 Su-Th
        elif part in _DAYS:
            days.append(_DAYS.index(part))
        else:
            return None
    return days


def _fmt(span: tuple[int, int] | None) -> list[str] | None:
    return None if span is None else [hhmm(span[0]), hhmm(span[1])]


def parse_opening_hours(text: str) -> dict[str, list[str] | None] | None:
    """把 OSM 的 opening_hours 轉成專案格式；看不懂的寫法回傳 None（改用估計值）。

    同一天分段營業時取最後一段（晚餐時段），寧可少排，也不要排到中間休息的時間。
    """
    text = (text or "").strip()
    if not text:
        return None
    if text == "24/7":
        return {"default": ["00:00", "24:00"]}
    week: list[tuple[int, int] | None] = [None] * 7  # OSM 沒寫到的星期視為公休
    for rule in filter(None, (r.strip() for r in text.split(";"))):
        if rule.startswith("PH"):
            continue
        m = _RULE.match(rule)
        if not m:
            return None
        days = _parse_days(m.group("days")) if m.group("days") else list(range(7))
        if days is None:
            return None
        if m.group("times") in ("off", "closed"):
            span = None
        else:
            o_h, o_m, c_h, c_m = map(int, _SPAN.findall(m.group("times"))[-1])
            if max(o_h, c_h) > 24 or max(o_m, c_m) > 59:
                return None
            open_, close = o_h * 60 + o_m, c_h * 60 + c_m
            if close <= open_:  # 營業到隔天凌晨
                close += 24 * 60
            span = (open_, close)
        for day in days:
            week[day] = span
    counts = Counter(week)
    default = max(counts, key=lambda s: (counts[s], s is not None))  # 一樣多時，營業的時段優先
    hours = {"default": _fmt(default)}
    for i, span in enumerate(week):
        if span != default:
            hours[WEEKDAY_KEYS[i]] = _fmt(span)
    return hours


def _cuisine_tokens(tags: dict) -> list[str]:
    return [t.strip().lower().replace(" ", "_") for t in tags.get("cuisine", "").split(";") if t.strip()]


def cuisine_category(tags: dict) -> str | None:
    amenity = tags.get("amenity")
    if amenity == "cafe":
        return "咖啡／甜點"
    if amenity in ("bar", "pub"):
        return "酒館"
    for token in _cuisine_tokens(tags):
        if token in CUISINES:
            return CUISINES[token][1]
    return None


def describe(tags: dict, category: str | None) -> str:
    amenity = tags.get("amenity", "restaurant")
    kind = KIND_WORDS.get(amenity, "餐廳")
    # 咖啡廳、酒館本身就說明了類型；餐廳、速食店前面加上料理類別
    parts = [kind if amenity in ("cafe", "bar", "pub") or not category else f"{category}{kind}"]
    dishes: list[str] = []
    for token in _cuisine_tokens(tags):
        word = CUISINES.get(token, ("",))[0]
        if word and not word.endswith("料理") and word not in dishes:
            dishes.append(word)
    if dishes:
        parts.append(f"供應{'、'.join(dishes)}")
    if tags.get("diet:vegetarian") in ("yes", "only") or tags.get("diet:vegan") in ("yes", "only"):
        parts.append("有素食選項")
    if tags.get("reservation") in RESERVABLE:
        parts.append("可訂位")
    if tags.get("takeaway") == "only":
        parts.append("以外帶為主")
    if tags.get("outdoor_seating") == "yes":
        parts.append("有戶外座位")
    text = "，".join(parts) + "。"
    if extra := tags.get("description", "").strip():
        text += extra if extra.endswith(("。", ".", "！", "!")) else f"{extra}。"
    return text


def place_hours(tags: dict) -> tuple[dict[str, list[str] | None], bool]:
    """營業時間與「是不是估計的」。"""
    if parsed := parse_opening_hours(tags.get("opening_hours", "")):
        return parsed, False
    open_, close = ESTIMATED_HOURS.get(tags.get("amenity", ""), ESTIMATED_HOURS["restaurant"])
    return {"default": [open_, close]}, True


def estimate_price(amenity: str, category: str | None) -> int:
    if amenity == "restaurant" and category in PRICEY_CATEGORIES:
        return 800
    return ESTIMATED_PRICE.get(amenity, ESTIMATED_PRICE["restaurant"])


def completeness(tags: dict) -> int:
    """資料完整度：有看得懂的營業時間 +2、有對得到的料理類別 +1。挑店時優先留資料完整的。"""
    has_hours = parse_opening_hours(tags.get("opening_hours", "")) is not None
    return 2 * has_hours + (cuisine_category(tags) is not None)
```

注意：`cuisine_category` 對 `cafe`／`bar`／`pub` 一律有類別，所以這三種的 `completeness` 至少是 1。

```python
def overpass_query(stations: list[Point]) -> str:
    amenity = "|".join(AMENITIES)
    lines = [
        f'  nwr["amenity"~"^({amenity})$"]["name"](around:{SEARCH_RADIUS_M},{p.lat:.6f},{p.lng:.6f});'
        for p in stations
    ]
    return "[out:json][timeout:60];\n(\n" + "\n".join(lines) + "\n);\nout center tags;"


def _point(element: dict) -> Point | None:
    if "lat" in element and "lon" in element:
        return Point(element["lat"], element["lon"])
    if center := element.get("center"):
        return Point(center["lat"], center["lon"])
    return None


def _address(tags: dict) -> str:
    if full := tags.get("addr:full"):
        return full
    return "".join(tags.get(k, "") for k in ("addr:city", "addr:district", "addr:street", "addr:housenumber"))


def to_record(element: dict, area_id: str) -> RestaurantRecord | None:
    tags = element.get("tags", {})
    point = _point(element)
    if point is None or not tags.get("name"):
        return None
    amenity = tags.get("amenity", "restaurant")
    category = cuisine_category(tags)
    hours, hours_estimated = place_hours(tags)
    osm_id = f"{element['type']}/{element['id']}"
    return RestaurantRecord(
        external_id=osm_id,
        name=tags["name"][:50],
        area_id=area_id,
        address=_address(tags)[:200],
        lat=point.lat,
        lng=point.lng,
        description=describe(tags, category),
        price_per_person=estimate_price(amenity, category),
        hours=hours,
        reservable=tags.get("reservation") in RESERVABLE,
        source_url=f"https://www.openstreetmap.org/{osm_id}",
        estimated=(["hours"] if hours_estimated else []) + ["price"],
    )


class OsmCollector(Collector):
    name = "osm"
    needs_coverage = True

    def __init__(self, coverage: Coverage, fetcher: Fetcher | None = None):
        self.coverage = coverage
        self.fetcher = fetcher or Fetcher(timeout=90)  # Overpass 忙的時候回應比較慢

    def collect(self, start: dt.date, days: int) -> Batch:
        restaurants: list[RestaurantRecord] = []
        done: list[str] = []
        failed = 0
        for area in self.coverage.areas:
            try:
                data = json.loads(self.fetcher.post(OVERPASS_URL, {"data": overpass_query(area.stations)}))
            except (FetchError, ValueError):
                failed += 1
                continue
            restaurants += self._pick(area, data.get("elements", []))
            done.append(area.id)
        if self.coverage.areas and not done:
            raise ScrapeError("Overpass API 所有商圈都查詢失敗")
        return Batch(restaurants=restaurants, restaurant_areas=done, stats={"failed_areas": failed})

    def _pick(self, area: AreaCoverage, elements: list[dict]) -> list[RestaurantRecord]:
        """一個商圈最多留 MAX_RESTAURANTS_PER_AREA 家：資料完整的優先，其次離影城近的。"""
        ranked = []
        seen: set[str] = set()
        for element in elements:
            point = _point(element)
            if point is None or self.coverage.area_for(point.lat, point.lng) != area.id:
                continue  # 離別的商圈更近（或超出範圍），交給那個商圈
            record = to_record(element, area.id)
            if record is None or record.external_id in seen:
                continue
            seen.add(record.external_id)
            tags = element.get("tags", {})
            ranked.append((-completeness(tags), area.walk_to_cinemas(point), record.name, record))
        ranked.sort(key=lambda item: item[:3])
        return [item[3] for item in ranked[:MAX_RESTAURANTS_PER_AREA]]
```

```python
"""背景自動更新爬蟲資料。

- 伺服器啟動後馬上檢查一次：OSM 超過 7 天沒成功更新、或開眼今天還沒成功抓過，就在背景執行。
- 之後每天 REFRESH_HOUR 點再檢查一次。
- 失敗只記錄在 scrape_runs，等下一輪再試；網頁的狀態列會顯示正在更新哪個來源。
"""
from __future__ import annotations

import datetime as dt
import logging
import threading
from collections.abc import Callable

from sqlalchemy import func, select
from sqlalchemy.orm import Session

from .collectors import get_collector
from .config import settings
from .db import SessionLocal
from .ingest import run_collector
from .models import ScrapeRun
from .timeutil import now_tw

log = logging.getLogger(__name__)

OSM_MAX_AGE = dt.timedelta(days=7)
SOURCES = ("osm", "atmovies")  # 先抓餐廳（約半分鐘），再抓場次


def due_sources(last_success: dict[str, dt.datetime | None], now: dt.datetime) -> list[str]:
    due = []
    osm = last_success.get("osm")
    if osm is None or now - osm >= OSM_MAX_AGE:
        due.append("osm")
    atmovies = last_success.get("atmovies")
    if atmovies is None or atmovies.date() < now.date():
        due.append("atmovies")
    return due


def next_check(now: dt.datetime, hour: int) -> dt.datetime:
    at = now.replace(hour=hour, minute=0, second=0, microsecond=0)
    return at if at > now else at + dt.timedelta(days=1)


def last_successes(session: Session) -> dict[str, dt.datetime | None]:
    rows = session.execute(
        select(ScrapeRun.source, func.max(ScrapeRun.finished_at))
        .where(ScrapeRun.status == "ok")
        .group_by(ScrapeRun.source)
    ).all()
    found = dict(rows)
    return {name: found.get(name) for name in SOURCES}


class Refresher:
    def __init__(
        self,
        session_factory: Callable = SessionLocal,
        hour: int | None = None,
        clock: Callable[[], dt.datetime] = now_tw,
    ):
        self.session_factory = session_factory
        self.hour = settings.refresh_hour if hour is None else hour
        self.clock = clock
        self.running: str | None = None  # 正在抓的來源
        self.last_errors: dict[str, str] = {}
        self._stop = threading.Event()
        self._thread: threading.Thread | None = None

    def start(self) -> None:
        if self._thread and self._thread.is_alive():
            return
        self._stop.clear()
        self._thread = threading.Thread(target=self._loop, name="refresher", daemon=True)
        self._thread.start()

    def stop(self) -> None:
        self._stop.set()
        if self._thread:
            self._thread.join(timeout=1)

    def run_due(self) -> list[str]:
        """檢查並執行該更新的來源，回傳這次執行了哪些。"""
        ran = []
        with self.session_factory() as session:
            for name in due_sources(last_successes(session), self.clock()):
                if self._stop.is_set():
                    break
                self.running = name
                try:
                    run = run_collector(session, get_collector(name, session))
                finally:
                    self.running = None
                ran.append(name)
                if run.status == "ok":
                    self.last_errors.pop(name, None)
                else:
                    self.last_errors[name] = run.error
                    log.warning("[%s] 更新失敗：%s", name, run.error)
        return ran

    def status(self) -> dict:
        return {"running": self.running, "errors": dict(self.last_errors), "auto": settings.auto_refresh}

    def _loop(self) -> None:
        while not self._stop.is_set():
            try:
                self.run_due()
            except Exception:  # 不要讓背景執行緒因為意外錯誤就停掉
                log.exception("自動更新發生錯誤")
            wait = (next_check(self.clock(), self.hour) - self.clock()).total_seconds()
            self._stop.wait(max(wait, 60))


refresher = Refresher()
```
